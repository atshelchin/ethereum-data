import Fuse from 'fuse.js'; // Load data const res = await fetch('./index/fuse-assets.json'); const { data, index } = await res.json(); // Create Fuse instance with pre-built index const fuseIndex = Fuse.parseIndex(index); const fuse = new Fuse(data, { keys: ['name', 'symbol'], threshold: 0.3 }, fuseIndex); // Search const results = fuse.search('usdt');
